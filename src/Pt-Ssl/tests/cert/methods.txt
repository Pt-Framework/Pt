openssl req -config ./openssl.cnf -newkey rsa:2048 -nodes -keyform PEM -keyout ca.key -x509 -days 36500 -extensions certauth -outform PEM -out ca.cer
openssl pkcs12 -export -inkey ca.key -in ca.cer -out ca.p12
openssl pkcs12 -export -inkey ca.key -in ca.cer -out ca-with-password.p12
openssl pkcs12 -in ca.p12 -out ca.pem -clcerts -nokeys
openssl x509 -in ca.pem -inform PEM -out ca.der -outform DER

openssl genrsa -out server.key 2048
openssl req -config ./openssl.cnf -new -key server.key -out server.req

openssl pkcs12 -export -inkey server.key -in server.cer -out server.p12

openssl x509 -in ca_1.pem -inform PEM -out ca_1.der -outform DER

openssl genrsa -out ca_2.key 2048
openssl req -config ./openssl_multi.cnf -new -key ca_2.key -out ca_2.req
openssl x509 -req -in ca_2.req -CA ca_1.cer -CAkey ca_1.key -set_serial 100 -extfile openssl_multi.cnf -extensions ca_2 -days 36400 -outform PEM -out ca_2.cer

cat root_ca.cer ca_1.cer ca_2.cer > ca_chain.cer

openssl genrsa -out server.key 2048
openssl req -config ./openssl_multi.cnf -new -key server.key -out server.req
